int getCalibratedMilliVolts() {
  int total = 0;
  for(int i = 0; i < 16; i++) {
    int raw = analogRead(ADC_PIN);
    total += (int)ADC_LUT[raw];
    delayMicroseconds(100);
  }
  int avg = total / 16;
  return map(avg, 0, 4095, 0, 3300) * DIVIDER_RATIO;
}

void startDischargeLoad(uint8_t dacValue) {

}
